// If you leave the conversation with Orpheus before making your decision, the player is informed about the Supreme Tadpole in their inventory.
IF

AND
GetHostCharacter((CHARACTER)_Player)
AND
DB_GlobalFlag((FLAG)LSK_KarsusRing_SupremeTadpoleGifted_fc94c99d-5bda-42f1-95ac-795b53a47ae0)

NOT DB_GlobalFlag((FLAG)END_IllithidOptions_State_ExitPortalAvailable_b97872a1-59cc-40cd-8963-e06be614adc0)

// Remove hidden tadpole for good once the exit portal is open.
IF
DialogEnded((DIALOGRESOURCE)END_IllithidOptions_OrpheusFreed_dd0df229-54bf-82cb-9869-a33edb46a57f,_InstanceID)
AND
DB_GlobalFlag((FLAG)LSK_KarsusRing_SupremeTadpoleGifted_fc94c99d-5bda-42f1-95ac-795b53a47ae0)
